-- =========================
-- 10) Update Orders.TotalAmount based on items
-- =========================
UPDATE o
SET o.TotalAmount = x.Total
FROM dbo.Orders o
JOIN (
    SELECT oi.OrderId, SUM(oi.Quantity * oi.UnitPrice) AS Total
    FROM dbo.OrderItems oi
    GROUP BY oi.OrderId
) x ON x.OrderId = o.Id;
